name: CI

on:
  push:
    branches: [main]
    paths:
      - 'src/**'
      - 'etl/**'
      - 'dev-tools/**'
      - 'boomtick-mcp/**'
      - 'tests/**'
      - 'scripts/**'
      - 'package.json'
      - 'pnpm-lock.yaml'
      - '.github/workflows/ci.yml'
      - '.github/actions/**'
      - 'AGENTS.md'
      - 'docs/github-workflows.md'
      - '*.config.ts'
      - 'tsconfig*.json'
  pull_request:
    branches: ['**']
    paths:
      - 'src/**'
      - 'etl/**'
      - 'dev-tools/**'
      - 'boomtick-mcp/**'
      - 'tests/**'
      - 'scripts/**'
      - 'package.json'
      - 'pnpm-lock.yaml'
      - '.github/workflows/ci.yml'
      - '.github/actions/**'
      - 'AGENTS.md'
      - 'docs/github-workflows.md'
      - '*.config.ts'
      - 'tsconfig*.json'
  workflow_dispatch:

permissions:
  contents: read
  actions: read
  statuses: write
  pull-requests: write
  models: read
  packages: read

concurrency:
  group: ${{ github.workflow }}-${{ github.ref }}
  cancel-in-progress: true

env:
  CI: true
  MAX_AI_REVIEWS: ${{ vars.MAX_AI_REVIEWS }}
  GITHUB_MODELS_MODEL: gpt-4o-mini
  ANY_COUNT_BASELINE: ${{ vars.ANY_COUNT_BASELINE }}
  BUNDLE_BASELINE_KB: ${{ vars.BUNDLE_BASELINE_KB }}
  AUDIT_BASELINE: ${{ vars.AUDIT_BASELINE }}

jobs:
  verify-changes:
    uses: ./.github/workflows/reusable-gate.yml
    permissions:
      pull-requests: write
      contents: write

  lint-typecheck:
    name: Lint & Type Check (${{ matrix.target }})
    needs: verify-changes
    if: needs.verify-changes.outputs.has_changes == 'true'
    runs-on: ubuntu-latest
    container:
      image: ghcr.io/arii/tech-dancer:latest
    strategy:
      fail-fast: false
      matrix:
        target: [root, boomtick-mcp]
    steps:
      - name: Configure Git Safe Directory
        run: git config --global --add safe.directory "$GITHUB_WORKSPACE"

      - name: Checkout
        uses: actions/checkout@v7
        with:
          fetch-depth: 0

      - name: Setup Node and pnpm
        uses: ./.github/actions/setup-node-pnpm

      - name: Setup Workspace
        uses: ./.github/actions/setup-workspace

      - name: Run Project Gate
        uses: ./.github/actions/run-project-gate
        with:
          target: ${{ matrix.target }}
          github_token: ${{ secrets.GITHUB_TOKEN }}

  audit:
    name: Anti-Pattern Audit
